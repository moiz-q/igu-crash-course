=== c-13-08 | exam | run | Find min and max via pointer parameters

--- solution
#include <stdio.h>

void minMax(int a, int b, int *mn, int *mx) {
    if (a < b) {
        *mn = a;
        *mx = b;
    } else {
        *mn = b;
        *mx = a;
    }
}

int main() {
    int x, y, lo, hi;
    scanf("%d %d", &x, &y);
    minMax(x, y, &lo, &hi);
    printf("min=%d max=%d\n", lo, hi);
    return 0;
}

- Assigns through `*mn = ...` and `*mx = ...`
- `main` passes `&lo, &hi`

--- solution
#include <stdio.h>

void stats(int *a, int n, int *sum, double *avg) {

    }
    *avg = (double)(*sum) / n;
